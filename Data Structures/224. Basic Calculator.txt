# 224. Basic Calculator 

# https://leetcode.com/problems/basic-calculator/?envType=study-plan-v2&envId=top-interview-150 

def calculate(s):
    
    d = {1: [0, "+"]} 
    idx = 0
    
    while idx < len(s):
        i = s[idx]
        
        if i == " ": 
            idx+=1 
            
        elif i in {"+", "-"}: 
            d[len(d)][1] = i 
            idx += 1
        
        elif i == "(": 
            d[len(d)+1] = [0, "+"]  
            idx+=1 
        
        elif i == ")": 
            temp = d[len(d)][0]
            del d[len(d)]
            
            if d[len(d)][1] == "+":
                d[len(d)][0] += temp 
            else: 
                d[len(d)][0] -= temp
            idx+=1 
            
        else: 
            r = idx
            while r<len(s) and ord(s[r])>= 48 and ord(s[r])<=57:
                r += 1 
            if d[len(d)][1] == "+":
                d[len(d)][0] += int(s[idx:r])  
            else: 
                d[len(d)][0] -= int(s[idx:r])
            idx = r  
            
    return d[1][0]


s = "1 + 1"
calculate(s)

s = " 2-1 + 2 "
calculate(s)

s = "(1+(4+5+2)-3)+(6+8)"
calculate(s)

s = "-20+(-4-3-2+9)-(-10-10)"
calculate(s)
