# 150. Evaluate Reverse Polish Notation

# https://leetcode.com/problems/evaluate-reverse-polish-notation/?envType=study-plan-v2&envId=top-interview-150 
tokens = ["2","1","+","3","*"] 

def evalRPN(tokens):
    if not tokens: return 0
    
    stack = [] 
    
    s = {"+", "-", "/", "*"}
    
    def DoMath(l, r, op): 
        l, r = int(l), int(r)
        
        if op == "+": 
            return l + r
        elif op == "-": 
            return l - r
        elif op == "/": 
            return round(l / r)
        elif op == "*": 
            return l * r
    
    for i in tokens: 
        if i in s: 
            r = stack.pop() 
            l = stack.pop() 
            
            stack.append(DoMath(l,r,i)) 
            
        else: 
            stack.append(i) 
    
    return stack[0]



tokens = ["2","1","+","3","*"] 
evalRPN(tokens)

tokens = ["4","13","5","/","+"]
evalRPN(tokens)

tokens = ["10","6","9","3","+","-11","*","/","*","17","+","5","+"]
evalRPN(tokens)
