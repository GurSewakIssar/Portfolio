    def __init__(self, val=None): 
        self.val = val 
        self.nextl = None
        
        
class TreeNode:

    while head:

                temp = temp.nextl 
            
            if tail.val.right: 
                temp.nextl = ListNodeL(tail.val.right)
                temp = temp.nextl 
            
            if prev: 
                prev.val.next = tail.val
            prev = tail 
            tail = tail.nextl 
        
        head = dummy1.nextl
