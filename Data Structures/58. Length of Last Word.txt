# 58. Length of Last Word 

def LengthOfLastWorld(s): 
    if not s: return
    
    i = len(s)-1
    while i >= 0 and not ((ord(s[i])>= 65 and ord(s[i])<= 90) or(ord(s[i])>= 97 and ord(s[i])<= 122)): 
        i -= 1 
        
    print(i)
    j = i
    while j>= 0 and ((ord(s[j])>= 65 and ord(s[j])<= 90) or(ord(s[j])>= 97 and ord(s[j])<= 122)):
        j -= 1  
    
    return i - j
