# 27. Remove Element 

def removeElement(nums, val):
    if not nums: return -1 
    
    def swap(l, r): 
        nums[l] = nums[r]  
        
    l,r = 0,0
    
    for i in range(len(nums)): 
        
        if l != r and nums[r] != val: 
            swap(l, r) 
            l += 1 
            r += 1

            
        elif nums[i] == val: 
            r += 1 
        else: 
            l += 1 
            r += 1 
    
    return len(nums) - l, nums

nums = [3,2,2,3] 
val = 3

nums = [0,1,2,2,3,0,4,2] 
val = 2

removeElement(nums,val)
