# 274. H-Index 

def hIndex(citations): 
    
    if not citations: return -1
    
    bucket = [0] * (len(citations) + 1) 
    
    for i in citations: 
        if i >= len(citations): 
            bucket[len(citations)] += 1 
        else: 
            bucket[i] += 1 
            
    count = 0 
    #print(bucket)
    
    for i in range(len(bucket)-1, -1, -1): 
        count += bucket[i] 
        if i <= count: return i  
    
citations = [3,0,6,1,5]
hIndex(citations)
citations = [1,3,1]
hIndex(citations)
