# 135. Candy 

def candy(ratings): 
    if not ratings: return  
    
    temp = [1] * len(ratings)
    
    for i in range(1, len(ratings)): 
        if ratings[i] > ratings[i-1]: 
            temp[i] += temp[i-1]
    
    prev = 1
    for i in range(len(ratings)-2, -1, -1): 
        if ratings[i] > ratings[i+1]: 
            prev +=1
            temp[i] = max(prev, temp[i])
        else: 
            prev = 1
            
    return sum(temp)
            
    
ratings = [1,0,2]
candy(ratings)

ratings = [1,2,2]
candy(ratings)

ratings = [5,4,3,2,1,3,2,1,2,3,2]
candy(ratings)
