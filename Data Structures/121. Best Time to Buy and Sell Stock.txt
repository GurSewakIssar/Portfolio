# 121. Best Time to Buy and Sell Stock

def maxProfit(nums):
    if not nums: return
    
    mn = float("infinity") 
    result = float("-infinity")  
    
    for i in nums: 
        mn = min(mn, i) 
        result = max(result, i - mn) 
        
    return result 


prices = [7,1,5,3,6,4]
prices = [7,6,4,3,1]

maxProfit(prices)
