# This is wrong, just change reverse loop to keep track of max of i+1 index and not i index
def maxProfit(prices): 
    profit = 0
    for i in range(1, len(prices)): 
       
        if prices[i]>prices[i-1]: 
            profit += prices[i]-prices[i-1]
        
    return profit
