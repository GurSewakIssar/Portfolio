# 55. jump Game

def canJump(nums): 
    if not nums: return -1 
    
    jumps = nums[0]
    
    for i in range(1, len(nums)): 
        
        if i == len(nums)-1: 
            return True 
        
        jumps = max(jumps-1, nums[i]) 
        
        if jumps <= 0: return False
    
    return False

nums = [2,3,1,1,4]
nums = [3,2,1,0,4]
nums = [3,3,1,0,1,2,0,1]

canJump(nums)
