# 53. Maximum Subarray

# https://leetcode.com/problems/maximum-subarray/?envType=study-plan-v2&envId=top-interview-150 

def maxSubArray(nums):
    
    temp = float("-inf") 
    result = float("-inf")
    
    for i in nums:
        if i > temp: 
            temp = i  
        else: 
            temp += i 
    
        result = max(result, temp) 
    
    return result
         

nums = [-2,1,-3,4,-1,2,1,-5,4] 
maxSubArray(nums)

nums = [1] 
maxSubArray(nums)

nums = [5,4,-1,7,8]
maxSubArray(nums)
