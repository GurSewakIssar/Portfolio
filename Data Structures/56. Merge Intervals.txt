# 56. Merge Intervals 

# https://leetcode.com/problems/merge-intervals/?envType=study-plan-v2&envId=top-interview-150 

def merge(intervals): 
    
    intervals = sorted(intervals) 
    l = intervals[0][0]
    r = intervals[0][1] 
    result = []
    
    for i in range(1, len(intervals)): 
        
        if intervals[i][0]<=r: 
            r = max(intervals[i][1], r) 
        else: 
            result.append([l, r]) 
            
            l = intervals[i][0]
            r = intervals[i][1]  
    
    result.append([l,r])
            
    return result


intervals = [[1,3],[2,6],[8,10],[15,18]]
merge(intervals)

intervals = [[1,4],[4,5]] 
merge(intervals)

intervals = [[4,7],[1,4]]
merge(intervals)           

intervals = [[4,7],[1,4], [6,12],[14,15], [13,21], [10,12]]
merge(intervals)
