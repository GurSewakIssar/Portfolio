# 13. Roman to Integer 

def RomanToInt(s): 
    if not s: return -1 
    
    d = {
        "I": 1, 
        "V": 5, 
        "X": 10, 
        "L": 50, 
        "C": 100, 
        "D": 500, 
        "M": 1000
        }
    
    result = 0 
    
    i = len(s)-1
    while i >= 0:
        
        c = d[s[i]]
        
        if i-1>=0 and s[i-1] == "I" and (s[i]=="V" or s[i] == "X"): 
            result += (c-1) 
            i -= 1
        elif i-1>=0 and s[i-1]  == "X" and (s[i]=="L" or s[i] == "C"): 
            result += (c-10)  
            i -= 1
        elif i-1>=0 and s[i-1]  == "C" and (s[i]=="D" or s[i] == "M"): 
            result += (c-100)  
            i -= 1
        else: 
            result += c 
        
        i -= 1
    
    return result  

s = "III"
RomanToInt(s)

s = "LVIII"
RomanToInt(s)

s = "MCMXCIV"
RomanToInt(s)
