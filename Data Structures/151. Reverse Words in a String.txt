# 151. Reverse Words in a String
# https://leetcode.com/problems/reverse-words-in-a-string/description/?envType=study-plan-v2&envId=top-interview-150

def reverseWords(s):
    if not s: return 
    
    i = len(s)-1
    result = ""
    while i >=0:
        
        if s[i] == " ":
            while i>= 0 and s[i] == " ": 
                i -= 1
        else:
            j = i
            while i >= 0 and s[i] != " ": 
                i -= 1  
    
            if not result: 
                result += s[i+1: j+1]
            else:
                result += " " + s[i+1: j+1]
 
    return result
            
        

s = "the sky is blue"
reverseWords(s)
    
s = "  hello world  "
reverseWords(s)
    
s = "a good   example" 
reverseWords(s)
