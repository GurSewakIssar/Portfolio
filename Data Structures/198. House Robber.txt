# 198. House Robber

# https://leetcode.com/problems/house-robber/?envType=study-plan-v2&envId=top-interview-150 

def rob(nums):
    if len(nums)<=2: return max(nums)
    
    for i in range(2,len(nums)): 
        if nums[i]+nums[i-2]>nums[i-1]: 
            nums[i] = nums[i] + nums[i-2] 
        else: 
            nums[i] = nums[i-1] 
    return nums[-1]

nums = [1,2,3,1] 
rob(nums)

nums = [2,7,9,3,1]
rob(nums)

nums = [1,2,5,6,8,1,3]
rob(nums)
