# 190. Reverse Bits 

# https://leetcode.com/problems/reverse-bits/description/?envType=study-plan-v2&envId=top-interview-150 

def reverseBits(n): 
    
    res = 0 
    i = 32
    while i: 
        res = res<<1
        res |= (n&1) 
        #res = res<<1 
        n = n>>1 
        i -=1
    
    return res 

n = 43261596 
reverseBits(n)
n = 2147483644
reverseBits(n)
